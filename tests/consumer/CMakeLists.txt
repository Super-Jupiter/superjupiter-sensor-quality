cmake_minimum_required(VERSION 3.16)
project(SuperJupiterSensorQualityConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_SENSOR_QUALITY_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_SENSOR_QUALITY_SOURCE_DIR}" sensor-quality)
  if(TARGET superjupiter-sensor-quality-tests OR TARGET superjupiter-sensor-quality-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterSensorQuality 0.2 CONFIG REQUIRED)
endif()
add_executable(superjupiter-sensor-quality-consumer main.cpp)
target_link_libraries(superjupiter-sensor-quality-consumer PRIVATE SuperJupiter::SensorQuality)
